/*Firma chce znaleźć produkty, które generują w danym kraju większy przychód niż przeciętny sprzedawany tam produkt.

Wynik powinien zawierać:
- country
- product_id
- product_name
- category
- units_sold
- orders_count
- total_revenue*/

with revenue_per_country as(
	select
	c.country,
	p.product_id,
	p.product_name,
	p.category,
	sum(oi.quantity) as units_sold,
	count(o.order_id) as orders_count,
	sum(oi.quantity * oi.unit_price) as total_revenue
	from course.customers c
	join course.orders o
	on c.customer_id = o.customer_id 
	join course.order_items oi
	on o.order_id = oi.order_id 
	join course.products p
	on oi.product_id = p.product_id 
	where o.status = 'paid'
	group by c.country,p.product_id
)
select *
from revenue_per_country
where total_revenue > (
	select avg(total_revenue)
	from revenue_per_country
)
