/*Zadanie: Klienci powyżej średniej sprzedaży
Pokaż klientów, których łączna wartość zamówień jest większa niż średnia łączna wartość zamówień per klient.
Wynik powinien zawierać:
- customer_id
- customer_name
- country
- orders_count
- total_revenue
Zasady:
- użyj tabel course.customers i course.orders,
- pokaż tylko klientów, którzy mają przynajmniej jedno zamówienie,
- orders_count to liczba zamówień klienta,
- total_revenue to suma orders.total_amount,
- użyj subquery do policzenia średniej łącznej wartości zamówień per klient,
- wynik posortuj po total_revenue malejąco.*/
select 
c.customer_id,
c.customer_name,
c.country,
count(o.order_id) as orders_count,
sum(o.total_amount) as total_revenue
from course.customers c
join course.orders o
on c.customer_id = o.customer_id 
group by o.customer_id,c.customer_id
having sum(o.total_amount) > (
select 
avg(customer_total) 
from (
	select sum(o.total_amount) as customer_total
	from course.orders o
	group by o.customer_id
))
order by total_revenue desc;


select 
avg(customer_total) 
from (
	select sum(o.total_amount) as customer_total
	from course.orders o
	group by o.customer_id
);


with paid_orders as (
select *
from course.orders 
where status = 'paid'
)
select *
from paid_orders;

with customer_totals as(
select 
sum(o.total_amount) as total_revenue,
o.customer_id
from course.orders o
group by o.customer_id),
average_customer_total as(
select
avg(total_revenue) as average_total_revenue 
from customer_totals
)
select 
c.customer_id,
c.customer_name,
c.country,
ct.total_revenue
from course.customers c
join customer_totals ct
on ct.customer_id = c.customer_id
join average_customer_total act
on ct.total_revenue > act.average_total_revenue
order by ct.total_revenue desc
